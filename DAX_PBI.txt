Total Respondents =
COUNTROWS(pept_clean)

#CREATE SATISFIED CUSTOMERS#
Satisfied Customers =
CALCULATE(
    COUNTROWS(pept_clean),
    pept_clean[pept_satisfaction] >= 4
)


#SATISFACTION RATE#
Satisfaction Rate =
DIVIDE(
    [Satisfied Customers],
    COUNT(pept_clean[pept_satisfaction]),
    0
)
